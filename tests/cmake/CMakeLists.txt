cmake_minimum_required(VERSION 3.16)
project(component_consumer LANGUAGES C)
enable_testing()
add_subdirectory(../.. component)
add_executable(consumer consumer.c)
target_link_libraries(consumer PRIVATE util_ringbuf)
set_target_properties(consumer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
add_test(NAME consumer COMMAND consumer)
# Optional freestanding Windows/LLVM host verification without a Windows SDK.
if(COMPONENT_FREESTANDING_TEST)
    set_target_properties(consumer PROPERTIES SUFFIX ".exe")
    target_sources(consumer PRIVATE ../runtime.c)
    target_compile_options(consumer PRIVATE -ffreestanding -fno-builtin -fno-stack-protector)
endif()
